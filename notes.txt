Init: Run npm init -y to create package.json.
Entry: Create index.js with a simple server.
Script: Add a start script to package.json.
Optional: Install Express with npm install express.
Run: Start with node index.js or npm start; use npx nodemon index.js for auto-reload.


in index.js 

import express from "express";
import cors from "cors";
import dotenv from "dotenv";
dotenv.config();

const app = express();
app.use(express.json());
app.use(express.urlencoded({ extended: true }));

app.use(
  cors({
    origin: ["http://localhost:5173"],
    credentials: true,
    methods: ["GET", "POST", "PUT", "DELETE"],
  })
);
